### Recommended Error Handling Flow

1. Business logic throws `AppError(code, type, message)`
2. Express/Fastify middleware catches → maps to HTTP status
3. Response: `{ "error": { "code", "type", "message" } }`
4. Client checks error code → looks up runbook

```typescript
// src/middleware/errorHandler.ts
import { AppError } from '../errors/AppError';

export function errorHandler(err: Error, req: Request, res: Response, next: NextFunction) {
  if (err instanceof AppError) {
    return res.status(err.httpStatus).json({
      error: {
        code: err.code,
        type: err.type,
        message: err.message
      }
    });
  }
  
  // Unknown error - log and return generic 500
  console.error('Unhandled error:', err);
  return res.status(500).json({
    error: {
      code: 50001,
      type: 'INTERNAL_ERROR',
      message: 'An unexpected error occurred'
    }
  });
}
```
